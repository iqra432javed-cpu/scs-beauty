// ── COUNTER ANIMATION (FIXED) ──────────────────────────────
// Root cause: IntersectionObserver never fired if section was
// already in viewport on load. Fix: dual trigger (immediate + observer).

function formatNumber(n) {
  if (n >= 1000) return (n / 1000).toFixed(0) + 'K';
  return n.toString();
}

function animateCounter(el) {
  if (el.dataset.animated) return; // prevent double-run
  el.dataset.animated = 'true';

  const target = parseInt(el.dataset.target, 10);
  const suffix = el.dataset.suffix || '';
  const duration = 2000;
  const startTime = performance.now();

  function step(currentTime) {
    const elapsed = currentTime - startTime;
    const progress = Math.min(elapsed / duration, 1);
    const eased = 1 - Math.pow(1 - progress, 3); // ease-out cubic
    const current = Math.floor(eased * target);
    el.textContent = formatNumber(current) + suffix;
    if (progress < 1) requestAnimationFrame(step);
    else el.textContent = formatNumber(target) + suffix;
  }

  requestAnimationFrame(step);
}

function initCounters() {
  const section = document.getElementById('countersSection');
  if (!section) return;
  const els = section.querySelectorAll('.counter-num[data-target]');

  // Check if already visible on page load
  const tryNow = () => {
    const rect = section.getBoundingClientRect();
    if (rect.top < window.innerHeight && rect.bottom > 0)
      els.forEach(animateCounter);
  };
  tryNow(); // immediate check

  const obs = new IntersectionObserver((entries) => {
    entries.forEach(e => {
      if (e.isIntersecting) { els.forEach(animateCounter); obs.disconnect(); }
    });
  }, { threshold: 0.2 });
  obs.observe(section);
}

document.addEventListener('DOMContentLoaded', initCounters);

function subscribeNewsletter() {
  const input = document.getElementById('newsletterEmail');
  const msg   = document.getElementById('newsletter-msg');
  if (!input || !msg) return;

  const email = input.value.trim();
  const valid = /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email);

  msg.style.color = '#B76E79';

  if (!email) {
    msg.textContent = 'Please enter your email address.';
    input.focus(); return;
  }
  if (!valid) {
    msg.textContent = 'Please enter a valid email address.';
    input.focus(); return;
  }

  input.value = '';
  msg.style.color = '#1D9E75';
  msg.textContent = "✓ You're in! Welcome to the SCS beauty community.";
  setTimeout(() => { msg.textContent = ''; }, 6000);
}

function injectProductSchema() {
  const schema = {
    "@context": "https://schema.org",
    "@type": "ItemList",
    "name": "SCS Beauty Products",
    "itemListElement": SCS_PRODUCTS.map((p, i) => ({
      "@type": "ListItem",
      "position": i + 1,
      "item": {
        "@type": "Product",
        "name": p.name,
        "description": p.desc,
        "brand": { "@type": "Brand", "name": "Success Chaser Stars" },
        "offers": {
          "@type": "Offer",
          "price": p.price,
          "priceCurrency": "PKR",
          "availability": "https://schema.org/InStock"
        }
      }
    }))
  };
  const el = document.createElement('script');
  el.type = 'application/ld+json';
  el.textContent = JSON.stringify(schema);
  document.head.appendChild(el);
}

// call it inside your DOMContentLoaded:
document.addEventListener('DOMContentLoaded', () => {
  renderProducts();
  injectProductSchema(); // ← ADD THIS LINE
